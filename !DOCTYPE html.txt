  <script>
    (function() {
      const problemInput = document.getElementById('problemInput');
      const doLabel = document.getElementById('doLabel');
      const notDoLabel = document.getElementById('notDoLabel');
      const resetBtn = document.getElementById('resetExampleBtn');
      const printBtn = document.getElementById('printPdfBtn');

      // Пример для сброса
      const defaultContent = {
        doWill: '• Новый опыт и профессиональный рост<br>• Возможность зарабатывать больше<br>• Расширение круга знакомств',
        doWont: '• Привычного коллектива и атмосферы<br>• Стабильности и уверенности в завтрашнем дне<br>• Накопленных бонусов (скидки, соцпакет)',
        notDoWill: '• Сохранение привычного уклада и коллектива<br>• Стабильный доход и график<br>• Отсутствие стресса при адаптации',
        notDoWont: '• Новых карьерных перспектив<br>• Повышения дохода<br>• Интересных вызовов и драйва'
      };

      function updateLabels(text) {
        const action = text.trim() || 'ЭТО';
        let displayAction = action.charAt(0).toUpperCase() + action.slice(1);
        doLabel.textContent = `Если я ${displayAction} СДЕЛАЮ…`;
        notDoLabel.textContent = `Если я ${displayAction} НЕ СДЕЛАЮ…`;
      }

      function resetExample() {
        document.getElementById('doWill').innerHTML = defaultContent.doWill;
        document.getElementById('doWont').innerHTML = defaultContent.doWont;
        document.getElementById('notDoWill').innerHTML = defaultContent.notDoWill;
        document.getElementById('notDoWont').innerHTML = defaultContent.notDoWont;
        problemInput.value = 'Стоит ли мне менять работу?';
        updateLabels(problemInput.value);
      }

      problemInput.addEventListener('input', function() {
        updateLabels(this.value);
      });

      resetBtn.addEventListener('click', resetExample);

      // Печать / PDF
      printBtn.addEventListener('click', function() {
        window.print();
      });

      // Инициализация
      updateLabels(problemInput.value);
    })();
  </script>
